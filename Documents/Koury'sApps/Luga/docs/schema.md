# Luga — Supabase database schema

## Conventions
- All primary keys: `uuid` generated by `gen_random_uuid()`
- All timestamps: `timestamptz` with `DEFAULT now()`
- All foreign keys have `ON DELETE CASCADE` unless noted
- Row Level Security (RLS) enabled on every table — no exceptions
- `updated_at` columns managed by `moddatetime` Supabase extension trigger

---

## users
```sql
CREATE TABLE users (
  id                  uuid PRIMARY KEY REFERENCES auth.users ON DELETE CASCADE,
  phone               text UNIQUE NOT NULL,
  name                text NOT NULL,
  display_name        text NOT NULL,
  photo_url           text,
  role                text NOT NULL DEFAULT 'sender', -- 'traveler' | 'sender' | 'both'
  kyc_level           int NOT NULL DEFAULT 0,          -- 0=none 1=basic 2=verified 3=trusted 4=elite
  kyc_status          text DEFAULT 'pending',          -- 'pending' | 'approved' | 'rejected'
  persona_inquiry_id  text,
  traveler_rating     numeric(3,2) DEFAULT 0,
  sender_rating       numeric(3,2) DEFAULT 0,
  trip_count          int DEFAULT 0,
  deal_count          int DEFAULT 0,
  completion_rate     numeric(5,2) DEFAULT 0,
  response_rate       numeric(5,2) DEFAULT 0,
  is_blocked          boolean DEFAULT false,
  referral_code       text UNIQUE,
  referred_by         uuid REFERENCES users,
  locale              text DEFAULT 'ar',
  created_at          timestamptz DEFAULT now(),
  updated_at          timestamptz DEFAULT now()
);
```

## trips
```sql
CREATE TABLE trips (
  id                  uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  traveler_id         uuid NOT NULL REFERENCES users ON DELETE CASCADE,
  origin_iata         text NOT NULL,        -- e.g. 'DXB', 'AUH', 'SHJ'
  destination_iata    text NOT NULL,        -- e.g. 'CAI', 'HBE'
  origin_city         text NOT NULL,        -- display name
  destination_city    text NOT NULL,
  flight_number       text,                 -- null = planned trip
  airline_code        text,
  departure_date      date NOT NULL,
  departure_time      time,
  avail_weight_kg     numeric(5,2) NOT NULL,
  remaining_weight_kg numeric(5,2) NOT NULL, -- updated on each booking
  handoff_type        text NOT NULL DEFAULT 'meetup', -- 'meetup' | 'courier'
  category_prices     jsonb DEFAULT '{}',   -- {electronics_mobile: 50, cosmetics: 15, ...}
  exclusions          text[] DEFAULT '{}',  -- category keys traveler won't carry
  status              text NOT NULL DEFAULT 'planned', -- 'planned'|'confirmed'|'active'|'completed'|'cancelled'
  is_round_trip       boolean DEFAULT false,
  return_date         date,
  note                text,
  created_at          timestamptz DEFAULT now(),
  updated_at          timestamptz DEFAULT now()
);

-- Index for discovery feed queries
CREATE INDEX trips_corridor_idx ON trips (origin_iata, destination_iata, departure_date, status);
```

## shipments
```sql
CREATE TABLE shipments (
  id                      uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  sender_id               uuid NOT NULL REFERENCES users ON DELETE CASCADE,
  origin_iata             text NOT NULL,
  destination_iata        text NOT NULL,

  destination_city        text NOT NULL,
  deadline_date           date NOT NULL,
  total_weight_g          int NOT NULL DEFAULT 0,    -- computed: sum of items
  total_declared_value_usd numeric(10,2) DEFAULT 0,  -- computed: sum of items
  suggested_reward_min_usd numeric(10,2),
  suggested_reward_max_usd numeric(10,2),
  delivery_pref           text NOT NULL DEFAULT 'meetup', -- 'meetup' | 'door'
  delivery_address        jsonb,                    -- {governorate, district, street, building, apt, landmark}
  status                  text NOT NULL DEFAULT 'open', -- 'open'|'matched'|'booked'|'in_transit'|'delivered'|'cancelled'
  note                    text,
  prohibited_ack          boolean NOT NULL DEFAULT false,
  created_at              timestamptz DEFAULT now(),
  updated_at              timestamptz DEFAULT now()
);

CREATE INDEX shipments_corridor_idx ON shipments (origin_iata, destination_iata, deadline_date, status);
```

## offers
```sql
CREATE TABLE offers (
  id              uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  trip_id         uuid NOT NULL REFERENCES trips ON DELETE CASCADE,
  shipment_id     uuid NOT NULL REFERENCES shipments ON DELETE CASCADE,
  traveler_id     uuid NOT NULL REFERENCES users,
  sender_id       uuid NOT NULL REFERENCES users,
  from_role       text NOT NULL,   -- 'traveler' | 'sender'
  amount_usd      numeric(10,2) NOT NULL,
  round_num       int NOT NULL DEFAULT 1,  -- 1, 2, or 3 only
  status          text NOT NULL DEFAULT 'pending', -- 'pending'|'countered'|'accepted'|'declined'|'expired'
  message         text,
  expires_at      timestamptz NOT NULL,    -- created_at + 6 hours
  created_at      timestamptz DEFAULT now()
);

-- Constraint: never allow round > 3
ALTER TABLE offers ADD CONSTRAINT max_three_rounds CHECK (round_num <= 3);
-- Constraint: amount must be positive
ALTER TABLE offers ADD CONSTRAINT positive_amount CHECK (amount_usd > 0);
```

## bookings
```sql
CREATE TABLE bookings (
  id                    uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  trip_id               uuid NOT NULL REFERENCES trips,
  shipment_id           uuid NOT NULL REFERENCES shipments,
  traveler_id           uuid NOT NULL REFERENCES users,
  sender_id             uuid NOT NULL REFERENCES users,
  agreed_price_usd      numeric(10,2) NOT NULL,
  sender_fee_usd        numeric(10,2) NOT NULL,  -- 10% of agreed
  traveler_commission_usd numeric(10,2) NOT NULL, -- 10% of agreed
  courier_addon_usd     numeric(10,2) DEFAULT 0,
  luga_revenue_usd      numeric(10,2) NOT NULL,
  protection_fund_usd   numeric(10,2) NOT NULL,  -- 20% of luga_revenue
  sender_total_usd      numeric(10,2) NOT NULL,  -- what sender pays
  traveler_payout_usd   numeric(10,2) NOT NULL,  -- what traveler receives
  stripe_payment_intent text NOT NULL,  -- Stripe PI ID — ALL payments via Stripe
  stripe_capture_id     text,           -- set when escrow captured on delivery
  status                text NOT NULL DEFAULT 'awaiting_payment',
  -- Status machine:
  -- awaiting_payment → escrow_held → pickup_confirmed → in_transit
  -- → delivery_confirmed → completed | disputed | refunded | cancelled
  pickup_photos         text[] DEFAULT '{}',
  delivery_photos       text[] DEFAULT '{}',
  pickup_confirmed_at   timestamptz,
  delivery_confirmed_at timestamptz,
  auto_release_at       timestamptz,  -- pickup_confirmed_at + 48h
  handoff_location      text,         -- meetup address or lat,lng
  delivery_address      jsonb,        -- for courier delivery
  phone_revealed_at     timestamptz,  -- when controlled phone reveal triggered
  shield_active         boolean DEFAULT true,
  cancellation_reason   text,
  cancelled_by          uuid REFERENCES users,
  stripe_payout_id      text,
  created_at            timestamptz DEFAULT now(),
  updated_at            timestamptz DEFAULT now()
);
```

## messages
```sql
CREATE TABLE messages (
  id              uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  booking_id      uuid NOT NULL REFERENCES bookings ON DELETE CASCADE,
  sender_id       uuid NOT NULL REFERENCES users,
  body            text NOT NULL,
  flagged         boolean DEFAULT false,
  flag_reason     text,    -- 'phone_number' | 'social_handle' | 'off_app_payment'
  is_system       boolean DEFAULT false,  -- system messages (status updates)
  created_at      timestamptz DEFAULT now()
);

-- Enable Supabase Realtime on this table
ALTER TABLE messages REPLICA IDENTITY FULL;
CREATE INDEX messages_booking_idx ON messages (booking_id, created_at);

-- Pre-payment message count view
CREATE VIEW pre_payment_message_counts AS
SELECT booking_id, COUNT(*) as count
FROM messages
WHERE is_system = false
  AND booking_id IN (SELECT id FROM bookings WHERE status = 'awaiting_payment')
GROUP BY booking_id;
```

## wallet_ledger
All amounts in USD. No EGP in the system — anti-inflation protection.
Sender payments go directly through Stripe (no wallet top-up needed).
This ledger tracks traveler earnings and platform transactions.
```sql
CREATE TABLE wallet_ledger (
  id              uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id         uuid NOT NULL REFERENCES users,
  type            text NOT NULL,  -- 'earning'|'payout'|'refund'|'fee'|'adjustment'
  amount_usd      numeric(12,2) NOT NULL,
  ref_id          uuid,           -- booking_id or payout reference
  stripe_txn_id   text,
  payout_method   text,           -- 'stripe_connect' | 'instapay' | null
  payout_status   text,           -- 'pending' | 'processing' | 'completed' | 'failed'
  idempotency_key text UNIQUE NOT NULL,
  description_ar  text,
  description_en  text,
  created_at      timestamptz DEFAULT now()
);

CREATE INDEX ledger_user_idx ON wallet_ledger (user_id, created_at DESC);
```

## RLS policies (critical — enable on all tables)
```sql
-- Users: read own row + public profiles
ALTER TABLE users ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Users can read own data" ON users FOR SELECT USING (auth.uid() = id);
CREATE POLICY "Users can update own data" ON users FOR UPDATE USING (auth.uid() = id);
CREATE POLICY "Public profiles readable" ON users FOR SELECT USING (true); -- name, photo, ratings only

-- Trips: travelers own their trips, everyone can read active ones
ALTER TABLE trips ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Traveler owns trip" ON trips USING (auth.uid() = traveler_id);
CREATE POLICY "Active trips public" ON trips FOR SELECT USING (status IN ('confirmed', 'planned'));

-- Bookings: only parties involved can see
ALTER TABLE bookings ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Booking parties only" ON bookings USING (
  auth.uid() = traveler_id OR auth.uid() = sender_id
);

-- Messages: only booking parties
ALTER TABLE messages ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Booking parties only" ON messages USING (
  booking_id IN (
    SELECT id FROM bookings
    WHERE traveler_id = auth.uid() OR sender_id = auth.uid()
  )
);

-- Wallet ledger: own transactions only
ALTER TABLE wallet_ledger ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Own ledger only" ON wallet_ledger USING (auth.uid() = user_id);
```
